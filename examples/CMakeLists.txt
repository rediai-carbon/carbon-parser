cmake_minimum_required(VERSION 3.16)
project(parser_installed_consumer LANGUAGES CXX)
if(NOT CMAKE_PREFIX_PATH)
  message(FATAL_ERROR "Set CMAKE_PREFIX_PATH to the installed carbon-parser prefix (vcpkg x64-linux-release root)")
endif()
list(GET CMAKE_PREFIX_PATH 0 CARBON_PARSER_ROOT)
get_filename_component(CARBON_PARSER_ROOT "${CARBON_PARSER_ROOT}" ABSOLUTE)
find_package(carbon-parser CONFIG REQUIRED)
add_executable(parser_installed_consumer installed_consumer.cpp)
# Installed CcpParser export omits INSTALL_INTERFACE includes (headers live under prefix/include).
target_include_directories(parser_installed_consumer PRIVATE "${CARBON_PARSER_ROOT}/include")
target_link_libraries(parser_installed_consumer PRIVATE CcpParser)
