cmake_minimum_required(VERSION 3.16)
project(parser_installed_consumer LANGUAGES CXX)
if(DEFINED CMAKE_PREFIX_PATH AND NOT CMAKE_PREFIX_PATH STREQUAL "")
  list(GET CMAKE_PREFIX_PATH 0 CARBON_PARSER_ROOT)
else()
  message(FATAL_ERROR "Set CMAKE_PREFIX_PATH to the installed carbon-parser prefix")
endif()
add_executable(parser_installed_consumer installed_consumer.cpp)
target_include_directories(parser_installed_consumer PRIVATE "${CARBON_PARSER_ROOT}/include")
target_link_libraries(parser_installed_consumer PRIVATE "${CARBON_PARSER_ROOT}/lib/CcpParser.a")
